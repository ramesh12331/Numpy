# ============================================================
# 1. WHAT IS NUMPY?
# ============================================================

# NumPy stands for Numerical Python.
#
# NumPy is a Python library mainly used for:
# - Numerical calculations
# - Arrays
# - Mathematical operations
# - Statistical calculations
# - Machine Learning
# - Scientific computing
#
# Main idea:
# NumPy provides powerful array-based tools
# for numerical computing in Python.


# ============================================================
# 2. WHY DO WE USE NUMPY?
# ============================================================

# Example:
# We have student marks.

marks = [70, 80, 90, 60, 85]

# We want to add 5 marks to every student.
#
# With a Python list, we normally need a loop.

new_marks = []

for mark in marks:
    new_marks.append(mark + 5)

print("Python List:")
print(new_marks)

# Output:
# [75, 85, 95, 65, 90]


# ============================================================
# USING NUMPY
# ============================================================

import numpy as np

marks = np.array([70, 80, 90, 60, 85])

# NumPy performs the operation element-by-element.

new_marks = marks + 5

print("\nNumPy Array:")
print(new_marks)

# Output:
# [75 85 95 65 90]


# ============================================================
# MATHEMATICAL DRY RUN
# ============================================================

# 70 + 5 = 75
# 80 + 5 = 85
# 90 + 5 = 95
# 60 + 5 = 65
# 85 + 5 = 90


# ============================================================
# 3. NUMPY VS PYTHON LIST
# ============================================================

# ---------------------------
# Python List
# ---------------------------

a = [1, 2, 3]
b = [4, 5, 6]

print("\nPython List Addition:")
print(a + b)

# Output:
# [1, 2, 3, 4, 5, 6]

# IMPORTANT:
# For Python lists:
# + means concatenation (joining).


# ---------------------------
# NumPy Array
# ---------------------------

a = np.array([1, 2, 3])
b = np.array([4, 5, 6])

print("\nNumPy Array Addition:")
print(a + b)

# Output:
# [5 7 9]

# NumPy performs element-wise addition:
#
# 1 + 4 = 5
# 2 + 5 = 7
# 3 + 6 = 9


# ============================================================
# 4. MULTIPLICATION DIFFERENCE
# ============================================================

# ---------------------------
# Python List
# ---------------------------

numbers = [1, 2, 3]

print("\nPython List Multiplication:")
print(numbers * 2)

# Output:
# [1, 2, 3, 1, 2, 3]

# Python repeats the list.


# ---------------------------
# NumPy Array
# ---------------------------

numbers = np.array([1, 2, 3])

print("\nNumPy Array Multiplication:")
print(numbers * 2)

# Output:
# [2 4 6]

# NumPy multiplies every element:
#
# 1 × 2 = 2
# 2 × 2 = 4
# 3 × 2 = 6


# ============================================================
# 5. PYTHON LIST VS NUMPY ARRAY
# ============================================================

# Python List:
# - General-purpose collection
# - Good for storing different types of data
# - Numerical operations usually require loops
# - Multidimensional numerical data is less convenient
#
# NumPy Array:
# - Designed for numerical computing
# - Direct element-wise operations
# - Supports multidimensional arrays
# - Provides many mathematical functions
# - Widely used in Data Science

# 13. REAL-WORLD EXAMPLE - SALES
# ============================================================

# Sales for five days.

sales = np.array([
    1000,
    1500,
    1200,
    1800,
    2000
])

# 10% discount means customer pays 90%.
#
# 100% - 10% = 90%
# 90% = 0.90

discounted_sales = sales * 0.90

print("\nOriginal Sales:")
print(sales)

print("\nAfter 10% Discount:")
print(discounted_sales)

# Output:
#
# Original Sales:
# [1000 1500 1200 1800 2000]
#
# After 10% Discount:
# [ 900. 1350. 1080. 1620. 1800.]


# ============================================================
# MATHEMATICAL DRY RUN
# ============================================================

# 1000 × 0.90 = 900
# 1500 × 0.90 = 1350
# 1200 × 0.90 = 1080
# 1800 × 0.90 = 1620
# 2000 × 0.90 = 1800
